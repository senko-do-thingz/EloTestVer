def build_leaderboard_embed(
    title: str,
    entries: list[tuple[int, str, str] | tuple[int, str, str, bool]],
    *,
    page: int | None = None,
    total_pages: int | None = None,
    note: str | None = None,
) -> tuple[discord.Embed, discord.File]:
    filename = "leaderboard.png"
    file = discord.File(
        render_leaderboard_image(entries, title=title, page=page, total_pages=total_pages),
        filename=filename,
    )
    embed = discord.Embed(color=discord.Color.blue())
    embed.set_image(url=f"attachment://{filename}")
    if note:
        embed.set_footer(text=note)
    return embed, file

                                async def _format_most_played_with(
                                    self, most_played_with, cursor, player_id: str
                                ):
                                    """Format most played with list with bold names, mentions, and games count."""
                                    if not most_played_with:
                                        return "Play at least 1 set to see your dream person"

                                    formatted_list = []
                                    for name in most_played_with:
                                        # Get user ID from database
                                        cursor.execute(
                                            "SELECT id FROM players WHERE name = ?",
                                            (name,),
                                        )
                                        row = cursor.fetchone()
                                        games = get_opponent_games(cursor, player_id, name)
                                        if row:
                                            user_id = int(row[0])
                                            mention = await self._safe_fetch_mention(
                                                user_id
                                            )
                                            formatted_list.append(
                                                f"**{name}** - {games} games ({mention})"
                                            )
                                        else:
                                            # If user not found in database, just show name in bold
                                            formatted_list.append(
                                                f"**{name}** - {games} games"
                                            )

                                    return "; ".join(formatted_list)

                            class PlayerSearchModal(
                                discord.ui.Modal, title="Search Player Stats"
                            ):
                                def __init__(self, view):
                                    super().__init__()
                                    self.view = view

                                player_name = discord.ui.TextInput(
                                    label="Player Name",
                                    placeholder="Enter player name to search...",
                                    required=True,
                                    max_length=50,
                                )

                                async def on_submit(
                                    self, interaction: discord.Interaction
                                ):
                                    search_name = self.player_name.value.strip().lower()

                                    try:
                                        # Read all stat files
                                        mpath = self.view.mpath
                                        file_map = {
                                            "elo": "data.txt",
                                            "wins": "w.txt",
                                            "losses": "l.txt",
                                            "win_rate": "wr.txt",
                                            "games_played_this_month": "games.txt",
                                        }

                                        stats_data = {}
                                        for category, filename in file_map.items():
                                            file_path = os.path.join(mpath, filename)
                                            if os.path.exists(file_path):
                                                with open(
                                                    file_path, "r", encoding="utf-8"
                                                ) as f:
                                                    lines = f.readlines()
                                                for line in lines:
                                                    stripped = line.strip()
                                                    if (
                                                        not stripped
                                                        or stripped.lower().endswith(
                                                            "leaderboard"
                                                        )
                                                    ):
                                                        continue
                                                    parsed = parse_leaderboard_line(
                                                        stripped,
                                                        win_rate=category == "win_rate",
                                                    )
                                                    if not parsed:
                                                        continue
                                                    name, value, _ = parsed
                                                    if search_name in name.lower():
                                                        if name not in stats_data:
                                                            stats_data[name] = {}
                                                        stats_data[name][category] = value

                                        if not stats_data:
                                            await interaction.response.send_message(
                                                embed=discord.Embed(
                                                    title="Player Not Found",
                                                    description=f"No player found with name containing `{search_name}`.",
                                                    color=discord.Color.red(),
                                                ),
                                                ephemeral=True,
                                            )
                                            return

                                        if len(stats_data) > 1:
                                            matches_list = "\n".join(
                                                [
                                                    f"- {name}"
                                                    for name in list(stats_data.keys())[
                                                        :10
                                                    ]
                                                ]
                                            )
                                            if len(stats_data) > 10:
                                                matches_list += f"\n... and {len(stats_data) - 10} more"
                                            await interaction.response.send_message(
                                                embed=discord.Embed(
                                                    title="Multiple Players Found",
                                                    description=f"Found {len(stats_data)} players:\n{matches_list}\n\nPlease be more specific.",
                                                    color=discord.Color.orange(),
                                                ),
                                                ephemeral=True,
                                            )
                                            return

                                        # Found exactly one player
                                        player_name = list(stats_data.keys())[0]
                                        player_stats = stats_data[player_name]

                                        def get_stat_value(category):
                                            value = player_stats.get(category)
                                            file_path = os.path.join(
                                                mpath, file_map[category]
                                            )
                                            if value is not None:
                                                return (
                                                    value,
                                                    True,
                                                )  # value exists and file exists
                                            if os.path.exists(file_path):
                                                return (
                                                    "-",
                                                    True,
                                                )  # value not found but file exists
                                            return "-", False  # file doesn't exist

                                        elo, elo_exists = get_stat_value("elo")
                                        wins, wins_exists = get_stat_value("wins")
                                        losses, losses_exists = get_stat_value("losses")
                                        games_played_this_month, games_exists = (
                                            get_stat_value("games_played_this_month")
                                        )
                                        win_rate, wr_exists = get_stat_value("win_rate")

                                        # Calculate ranks from files
                                        def get_rank_for_stat(category, value):
                                            file_path = os.path.join(
                                                mpath, file_map[category]
                                            )
                                            if not os.path.exists(file_path):
                                                return None
                                            with open(
                                                file_path, "r", encoding="utf-8"
                                            ) as f:
                                                lines = f.readlines()
                                            all_values = []
                                            for line in lines:
                                                stripped = line.strip()
                                                if (
                                                    not stripped
                                                    or stripped.lower().endswith(
                                                        "leaderboard"
                                                    )
                                                ):
                                                    continue
                                                parsed = parse_leaderboard_line(
                                                    stripped,
                                                    win_rate=category == "win_rate",
                                                )
                                                if parsed:
                                                    all_values.append(parsed[1])  # value
                                            if not all_values:
                                                return None
                                            all_values.sort(reverse=True)
                                            return (
                                                sum(1 for v in all_values if v > value)
                                                + 1
                                            )

                                        elo_rank = (
                                            get_rank_for_stat("elo", elo)
                                            if elo != "-"
                                            else None
                                        )
                                        wins_rank = (
                                            get_rank_for_stat("wins", wins)
                                            if wins != "-"
                                            else None
                                        )
                                        losses_rank = (
                                            get_rank_for_stat("losses", losses)
                                            if losses != "-"
                                            else None
                                        )
                                        win_rate_rank = (
                                            get_rank_for_stat("win_rate", win_rate)
                                            if win_rate != "-"
                                            else None
                                        )

                                        # Build description with conditional rank display
                                        stat_lines = []
                                        stat_lines.append(
                                            f"**🔥 Elo:** {elo}"
                                            + (
                                                f" (#{elo_rank})"
                                                if elo_rank is not None
                                                else ""
                                            )
                                        )
                                        stat_lines.append(
                                            f"**🏆 Wins:** {wins}"
                                            + (
                                                f" (#{wins_rank})"
                                                if wins_rank is not None
                                                else ""
                                            )
                                        )
                                        stat_lines.append(
                                            f"**💀 Losses:** {losses}"
                                            + (
                                                f" (#{losses_rank})"
                                                if losses_rank is not None
                                                else ""
                                            )
                                        )
                                        # Only show monthly games played if not overall month
                                        if self.view.month.lower() != "overall":
                                            monthly_rank = (
                                                get_rank_for_stat(
                                                    "games_played_this_month",
                                                    games_played_this_month,
                                                )
                                                if games_played_this_month != "-"
                                                else None
                                            )
                                            stat_lines.append(
                                                f"**🎮 Monthly Games Played:** {games_played_this_month}"
                                                + (
                                                    f" (#{monthly_rank})"
                                                    if monthly_rank is not None
                                                    else ""
                                                )
                                            )
                                        stat_lines.append(
                                            f"**📊 Win Rate:** {win_rate:.2f}%"
                                            if win_rate != "-"
                                            else f"**📊 Win Rate:** {win_rate}"
                                        )
                                        if win_rate_rank is not None:
                                            stat_lines[-1] += f" (#{win_rate_rank})"

                                        # Get Most Played With from database
                                        try:
                                            with db_connection(
                                                lb_guild_id
                                            ) as (conn, cursor):
                                                cursor.execute(
                                                    "SELECT id FROM players WHERE name = ?",
                                                    (player_name,),
                                                )
                                                id_row = cursor.fetchone()
                                                if not id_row:
                                                    most_played_text = "-"
                                                else:
                                                    player_id = str(id_row[0])
                                                    most_played_with = get_most_played_with(
                                                        cursor, player_id
                                                    )
                                                    if not most_played_with:
                                                        most_played_text = "-"
                                                    else:
                                                        most_played_text = await self.view._format_most_played_with(
                                                            most_played_with,
                                                            cursor,
                                                            player_id,
                                                        )
                                            stat_lines.append(
                                                f"**👥 Most Played Against:** {most_played_text}"
                                            )
                                        except Exception:
                                            # If database query fails, just skip this stat
                                            pass

                                        embed = discord.Embed(
                                            title=f"All Stats for {player_name}",
                                            description="\n".join(stat_lines),
                                            color=discord.Color.green(),
                                        )
                                        embed.set_footer(
                                            text=f"From {self.view.season} / {self.view.month}"
                                        )

                                        # Send followup message with stats
                                        await interaction.response.send_message(
                                            embed=embed, ephemeral=True
                                        )

                                    except Exception as e:
                                        await interaction.response.send_message(
                                            embed=discord.Embed(
                                                title="Error",
                                                description=f"An error occurred: {str(e)}",
                                                color=discord.Color.red(),
                                            ),
                                            ephemeral=True,
                                        )

                            lb_view = LBView(
                                self.season,
                                month,
                                mpath,
                                months,
                                category="elo",
                                bot=bot,
                            )
                            await lb_view.update(interaction)

                        except Exception as e:
                            await interaction.response.edit_message(
                                embed=discord.Embed(
                                    title="Error",
                                    description=str(e),
                                    color=discord.Color.red(),
                                ),
                                view=None,
                            )

                await interaction.response.edit_message(
                    embed=discord.Embed(
                        title=f"Select a Month in {season}",
                        description="Choose a month to view its leaderboard.",
                        color=discord.Color.blue(),
                    ),
                    view=MonthView(),
                )

        await ctx.send(
            embed=discord.Embed(
                title="Select a Season",
                description="Choose a season to view its leaderboards.",
                color=discord.Color.blue(),
            ),
            view=SeasonView(),
        )

    @commands.command(name="lbs_id", hidden=True)
    async def lbs_id(self, ctx: commands.Context, server_id: int):
        """View archived seasonal leaderboards for another server (restricted guild only)."""
        if ctx.guild is None or ctx.guild.id != LB_ID_ALLOWED_GUILD_ID:
            await ctx.send(
                embed=discord.Embed(
                    title="Not Available",
                    description="This command is not available in this server.",
                    color=discord.Color.red(),
                )
            )
            return

        base = get_lb_root(server_id)
        if not os.path.isdir(base):
            await ctx.send(
                embed=discord.Embed(
                    title="Leaderboard Archive Not Found",
                    description=(
                        f"No leaderboard archive found for server ID **{server_id}** "
                        f"(`{base.replace(os.sep, '/')}/`). "
                        f"Expected `lb/{server_id}/` or `lb/{server_id}_<note>/`."
                    ),
                    color=discord.Color.red(),
                )
            )
            return

        await self._open_lbs_picker(ctx, server_id)
