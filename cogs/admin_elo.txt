class AdminEloCog(commands.Cog):
    def __init__(self, bot: commands.Bot):
        self.bot = bot

    async def cog_command_error(self, ctx: commands.Context, error: Exception):
        if isinstance(error, commands.CheckFailure):
            await ctx.send(
                embed=discord.Embed(
                    title="Permission Denied",
                    description="You do not have permission to use this admin command.",
                    color=discord.Color.red(),
                )
            )
            return
        raise error

    async def _remove_bot_reactions(self, message: discord.Message) -> None:
        bot_user = getattr(self.bot, "user", None)
        if not bot_user:
            return
        for emoji in (YES_EMOJI, NO_EMOJI):
            try:
                await message.remove_reaction(emoji, bot_user)
            except Exception:
                pass

    async def apply_elo_decay(self, guild_id: int):
        with db_connection(guild_id) as (conn, cursor):
            cursor.execute("SELECT id, elo, games_played_this_month FROM players")
            players = cursor.fetchall()

            for player_id, elo, games_played_this_month in players:
                # Skip players who played 0 monthly games
                if games_played_this_month == 0:
                    continue

                diff = elo - 500
                if abs(diff) % 2 == 1:
                    elo = 500 + math.ceil(diff / 2)
                else:
                    elo = 500 + diff // 2

                if games_played_this_month < 15:
                    elo -= (15 - games_played_this_month) * 2

                cursor.execute(
                    "UPDATE players SET elo = ? WHERE id = ?",
                    (max(elo, 0), player_id),
                )

            conn.commit()
            lb_cache_invalidate(ctx.guild.id)
            lb_cache_rebuild(self.bot, ctx.guild.id)

    @commands.command()
    @commands.check(is_admin)
    async def decay(self, ctx: commands.Context):
        embed = discord.Embed(
            title="Elo Decay Confirmation",
            description=(
                f"**This function affects all player stats**, Are you sure you want to apply Elo decay? React with {YES_EMOJI} to confirm or {NO_EMOJI} to cancel within 5 minutes."
            ),
            color=discord.Color.orange(),
        )
        message = await ctx.send(embed=embed)
        await message.add_reaction(YES_EMOJI)
        await message.add_reaction(NO_EMOJI)

        def check(reaction, user):
            return (
                user == ctx.author
                and str(reaction.emoji) in [YES_EMOJI, NO_EMOJI]
                and reaction.message.id == message.id
            )

        try:
            reaction, user = await self.bot.wait_for(
                "reaction_add", timeout=300.0, check=check
            )
            if str(reaction.emoji) == YES_EMOJI:
                await self.apply_elo_decay(ctx.guild.id)
                with db_connection(ctx.guild.id) as (conn, cursor):
                    cursor.execute("UPDATE players SET games_played_this_month=0")
                    conn.commit()
                lb_cache_invalidate(ctx.guild.id)
                lb_cache_rebuild(self.bot, ctx.guild.id)
                await sync_league_roles(ctx.guild)
                embed.title = "Elo Decay Applied"
                embed.description = "Elo decay has been applied successfully, and monthly stats have been reset!"
                embed.color = discord.Color.green()
                await bot_log(
                    ctx.bot,
                    ctx.guild.id,
                    "Command used: -decay",
                    level="WARNING",
                    fields={
                        "User": f"{ctx.author.mention} ({ctx.author.id})",
                        "Input": f"`{ctx.message.content[:200]}`",
                    },
                )
            else:
                embed.title = "Elo Decay Canceled"
                embed.description = "Elo decay has been canceled."
                embed.color = discord.Color.red()

            await self._remove_bot_reactions(message)
            await message.edit(embed=embed)
        except asyncio.TimeoutError:
            await self._remove_bot_reactions(message)
            await message.edit(embed=discord.Embed(
                title="⌛ Timeout",
                description="Elo decay confirmation timed out. Please try again.",
                color=discord.Color.red(),
            ))

    async def db_d(self, ctx: commands.Context, server_id: int, record_id: int):
        if not db_exists(server_id):
            await ctx.send(
                embed=discord.Embed(
                    title="Database Not Found",
                    description=(
                        f"No database found for server ID {server_id} "
                        f"(`{get_db_path(server_id).replace(os.sep, '/')}`)."
                    ),
                    color=discord.Color.red(),
                )
            )
            return
        with db_connection(server_id) as (conn, cursor):
            cursor.execute("SELECT * FROM players WHERE id=?", (record_id,))
            record = cursor.fetchone()
        if not record:
            await ctx.send(
                embed=discord.Embed(
                    title="Record Not Found",
                    description=(
                        f"No record found with ID {record_id} in the database for server ID {server_id}."
                    ),
                    color=discord.Color.red(),
                )
            )
            return
        embed = discord.Embed(
            title="Confirm Deletion",
            description=(
                f"Are you sure you want to delete the record with ID {record_id} from the database for server ID {server_id}? React with {YES_EMOJI} to confirm or {NO_EMOJI} to cancel."
            ),
            color=discord.Color.orange(),
        )
        message = await ctx.send(embed=embed)
        await message.add_reaction(YES_EMOJI)
        await message.add_reaction(NO_EMOJI)

        def check(reaction, user):
            return (
                user == ctx.author
                and str(reaction.emoji) in [YES_EMOJI, NO_EMOJI]
                and reaction.message.id == message.id
            )

        try:
            reaction, _ = await self.bot.wait_for(
                "reaction_add", timeout=300.0, check=check
            )
            if str(reaction.emoji) == YES_EMOJI:
                with db_connection(server_id) as (conn, cursor):
                    cursor.execute("DELETE FROM players WHERE id=?", (record_id,))
                    conn.commit()
                embed = discord.Embed(
                    title="Record Deleted",
                    description=(
                        f"The record with ID {record_id} has been deleted from the database for server ID {server_id}."
                    ),
                    color=discord.Color.green(),
                )
            else:
                embed = discord.Embed(
                    title="Deletion Canceled",
                    description=(
                        f"The deletion of the record with ID {record_id} has been canceled."
                    ),
                    color=discord.Color.red(),
                )
            await self._remove_bot_reactions(message)
            await message.edit(embed=embed)
        except asyncio.TimeoutError:
            await self._remove_bot_reactions(message)
            await message.edit(embed=discord.Embed(
                title="Timeout",
                description="Deletion confirmation timed out. Please try again.",
                color=discord.Color.red(),
            ))
